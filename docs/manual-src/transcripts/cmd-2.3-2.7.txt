##### 2.3 外網下載:tar.gz(tar.gz 安裝方式用;本次只取 CP 8.3.2 與 JDK,不取升級演練用的舊版與 C3 tar)
mkdir -p /out/tar && cd /out/tar && curl -fsSL --retry 3 -o confluent-8.3.2.tar.gz.part https://packages.confluent.io/archive/8.3/confluent-8.3.2.tar.gz && mv confluent-8.3.2.tar.gz.part confluent-8.3.2.tar.gz && ls -l /out/tar
curl -fsSL 'https://api.adoptium.net/v3/assets/latest/21/hotspot?os=linux&architecture=x64&image_type=jdk&vendor=eclipse' | python3 -c 'import json,sys; a=json.load(sys.stdin)[0]; p=a["binary"]["package"]; print(p["link"]); print(p["name"]); print(p["checksum"]); print(a["release_name"])' | tee /tmp/jdk.txt
cd /out/tar && link=$(sed -n 1p /tmp/jdk.txt) && name=$(sed -n 2p /tmp/jdk.txt) && sum=$(sed -n 3p /tmp/jdk.txt) && curl -fsSL --retry 3 -o "$name" "$link" && echo "$sum  $name" | sha256sum -c && ls -l /out/tar
##### 2.4 外網下載:JMX exporter(以 Maven Central 的 sha1 驗證)
mkdir -p /out/tools && cd /out/tools && curl -fsSL --retry 3 -o jmx_prometheus_javaagent-1.0.1.jar https://repo1.maven.org/maven2/io/prometheus/jmx/jmx_prometheus_javaagent/1.0.1/jmx_prometheus_javaagent-1.0.1.jar && echo "official $(curl -fsSL https://repo1.maven.org/maven2/io/prometheus/jmx/jmx_prometheus_javaagent/1.0.1/jmx_prometheus_javaagent-1.0.1.jar.sha1)" && echo "local    $(sha1sum jmx_prometheus_javaagent-1.0.1.jar | cut -d' ' -f1)" && stat -c '%s bytes' jmx_prometheus_javaagent-1.0.1.jar
##### 2.5 打包 OP menu(可重現的 tarball,以內容雜湊當版本)
find /src/opmenu/opmenu.sh /src/opmenu/lib /src/opmenu/README.md /src/opmenu/opmenu.conf.example -type f -print0 | xargs -0 perl -ne 'if (/\r$/) { print "CRLF: $ARGV\n"; exit 1 }' && echo no-crlf
stage=$(mktemp -d) && mkdir -p "$stage/opmenu" && cp -r /src/opmenu/opmenu.sh /src/opmenu/lib /src/opmenu/README.md /src/opmenu/opmenu.conf.example "$stage/opmenu/" && { [ -f /src/opmenu/test-host-gate.sh ] && cp /src/opmenu/test-host-gate.sh "$stage/opmenu/" || true; } && tar --sort=name --mtime='2026-01-01 00:00:00' --owner=0 --group=0 --numeric-owner -C "$stage" -cf "$stage/opmenu.tar" opmenu && gzip -n -9 "$stage/opmenu.tar" && sha=$(sha256sum "$stage/opmenu.tar.gz" | awk '{print $1}') && ver="$(date +%Y%m%d).${sha:0:8}" && install -m 0644 "$stage/opmenu.tar.gz" "/out/tools/opmenu-$ver.tar.gz" && rm -rf "$stage" && echo "OPMENU_VERSION=$ver" && tar -tzf /out/tools/opmenu-$ver.tar.gz | head -20
##### 2.6 產生 SHA256SUMS(相對路徑;bootstrap/ 也列入)
cd /out && find rpm tar tools gpg-keys bootstrap -type f 2>/dev/null | LC_ALL=C sort | xargs -r sha256sum > SHA256SUMS.tmp && mv SHA256SUMS.tmp SHA256SUMS && wc -l SHA256SUMS && head -3 SHA256SUMS
cd /out && for f in rpm/*/*.rpm; do printf '%s|%s|%s\n' "$(basename "$(dirname "$f")")" "$(rpm -qp --qf '%{NAME}|%{VERSION}-%{RELEASE}|%{VENDOR}' "$f" 2>/dev/null)" "$(sha256sum "$f" | awk '{print $1}')"; done > manifest-rpm.txt && sha256sum tar/*.tar.gz > manifest-tar.txt && wc -l manifest-rpm.txt manifest-tar.txt && grep '|confluent-server|' manifest-rpm.txt
##### 2.7 外網驗證閘門(checksum + GPG + 清單),產出可附在變更單的報告
cd /out && mkdir -p report && r=report/verify-$(date +%Y%m%d-%H%M%S).txt && { echo "== sha256 =="; sha256sum -c --quiet SHA256SUMS && echo "all $(wc -l <SHA256SUMS) files match SHA256SUMS"; echo "== gpg =="; for k in gpg-keys/*.asc; do rpm --import "$k"; done; find rpm bootstrap -name '*.rpm' -print0 | xargs -0 rpm -K | grep -v 'digests signatures OK' | sed 's/^/BAD: /'; echo "all $(find rpm bootstrap -name '*.rpm' | wc -l) RPMs checked"; echo "== package inventory =="; cut -d'|' -f2,3,4 manifest-rpm.txt | grep -i confluent | column -t -s'|'; } | tee "$r"; echo "report: $r"
