{{

##### 4.1.2 OU、群組(ad01,root)
{{
ssh ad01.poc.lab 'sudo bash -s' <<'EOF'
set -e
LDB=/var/lib/samba/private/sam.ldb
for ou in "OU=Confluent" "OU=Users,OU=Confluent" "OU=Groups,OU=Confluent" "OU=Service,OU=Confluent"; do
  ldbsearch -H $LDB -b "$ou,DC=corp,DC=poc,DC=lab" -s base dn 2>/dev/null | grep -q '^dn:' && echo "ok  $ou" || samba-tool ou create "$ou"
done
for g in kafka-admins kafka-operators kafka-devs; do
  ldbsearch -H $LDB "(&(objectClass=group)(sAMAccountName=$g))" dn | grep -q '^dn:' && echo "ok  $g" || samba-tool group add "$g" --groupou="OU=Groups,OU=Confluent"
done
EOF
}}
##### 4.1.2 服務帳號與測試使用者:密碼從 ops02 的 0600 密碼檔經 stdin 送給 samba-tool(不在指令列、不在 ps;samba-tool 非終端機時從 stdin 讀密碼)
{{
. ~/manual.env; set -a; . $S/secrets.env; set +a
mkuser() {  # <user> <ou> <password> [group]
  if ssh ad01.poc.lab "sudo ldbsearch -H /var/lib/samba/private/sam.ldb '(&(objectClass=user)(sAMAccountName=$1))' dn" | grep -q '^dn:'; then echo "ok  $1"; else
    printf '%s\n%s\n' "$3" "$3" | ssh ad01.poc.lab "sudo samba-tool user create $1 --userou='$2' 2>&1 | tail -1"
    ssh ad01.poc.lab "sudo samba-tool user setexpiry $1 --noexpiry >/dev/null"
  fi
  [ -n "${4:-}" ] && ssh ad01.poc.lab "sudo samba-tool group addmembers $4 $1 2>&1 | tail -1" || true
}
mkuser svc-mds-bind "OU=Service,OU=Confluent" "$LDAP_BIND_PASSWORD"
mkuser svc-rest "OU=Service,OU=Confluent" "$SVC_REST_PASSWORD"
mkuser svc-c3 "OU=Service,OU=Confluent" "$SVC_C3_PASSWORD"
mkuser svc-cpctl "OU=Service,OU=Confluent" "$SVC_CPCTL_PASSWORD"
mkuser alice "OU=Users,OU=Confluent" "$HUMAN_DEFAULT_PASSWORD" kafka-admins
mkuser bob "OU=Users,OU=Confluent" "$HUMAN_DEFAULT_PASSWORD" kafka-operators
mkuser carol "OU=Users,OU=Confluent" "$HUMAN_DEFAULT_PASSWORD" kafka-devs
}}
##### 4.1 驗證:以 svc-mds-bind 經 LDAPS bind(與 MDS 做的事相同);密碼放 0600 檔案用 -y,不用 -w
{{
. ~/manual.env; set -a; . $S/secrets.env; set +a
ssh ad01.poc.lab "sudo samba-tool ou list; for g in kafka-admins kafka-operators kafka-devs; do echo \"\$g: \$(sudo samba-tool group listmembers \$g | tr '\n' ' ')\"; done"
install -m 0600 /dev/null ~/.bindpw && printf '%s' "$LDAP_BIND_PASSWORD" > ~/.bindpw
LDAPTLS_CACERT=$PKI/ca.pem ldapsearch -LLL -x -o nettimeout=5 -H ldaps://ad01.poc.lab:636 -D "CN=svc-mds-bind,OU=Service,OU=Confluent,DC=corp,DC=poc,DC=lab" -y ~/.bindpw -b "OU=Confluent,DC=corp,DC=poc,DC=lab" "(objectClass=user)" sAMAccountName | grep sAMAccountName
LDAPTLS_CACERT=$PKI/ca.pem ldapsearch -LLL -x -o nettimeout=5 -H ldaps://ad01.poc.lab:636 -D "CN=svc-mds-bind,OU=Service,OU=Confluent,DC=corp,DC=poc,DC=lab" -y ~/.bindpw -b "OU=Groups,OU=Confluent,DC=corp,DC=poc,DC=lab" "(cn=kafka-admins)" member
rm -f ~/.bindpw
}}
