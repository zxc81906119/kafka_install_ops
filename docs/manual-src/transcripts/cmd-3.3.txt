##### 3.3.1.3 步驟 02:依角色安裝精確版本 + versionlock(ops02 先做;cpadmin + sudo)
{{
install_pkgs() {   # <host> <package-version...>
  local h=$1; shift
  ssh $h.poc.lab "PKGS='$*' bash -s" <<'EOS'
set -e
rpm -q --qf '%{NAME} %{VERSION}\n' temurin-21-jdk confluent-server confluent-kafka-rest confluent-control-center-next-gen 2>/dev/null | grep -v 'not installed' || echo "no Java/Confluent package installed yet"
sudo dnf -y -q --disablerepo='*' --enablerepo='cp-internal-*' --setopt=install_weak_deps=False install $PKGS >/dev/null
for p in $PKGS; do sudo dnf -q versionlock add "$p" >/dev/null; done
echo "$(hostname): $(rpm -q --qf '%{NAME}-%{VERSION} ' $(for p in $PKGS; do echo "${p%-*}"; done))"
EOS
}
J=temurin-21-jdk-21.0.12.1.0.1
install_pkgs ops02 $J confluent-server-8.3.2 confluent-security-8.3.2 confluent-cli-8.3.2
for h in kc01 kc02 kc03 kb01 kb02 kb03 kb04 jump01; do install_pkgs $h $J confluent-server-8.3.2 confluent-security-8.3.2; done
for h in rest01 rest02; do install_pkgs $h $J confluent-kafka-rest-8.3.2 confluent-security-8.3.2; done
c3os=$(for p in cyrus-sasl openssl-devel; do ls /srv/repo/rpm/os-deps/$p-[0-9]*.rpm | sort -V | tail -1 | xargs rpm -qp --qf "%{NAME}-%{VERSION}\n"; done | tr '\n' ' ')
echo "c3 OS dependencies from the internal repo: $c3os"
install_pkgs c3-01 $J confluent-control-center-next-gen-2.6.1 $c3os
}}
