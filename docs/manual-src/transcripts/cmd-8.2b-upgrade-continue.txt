##### 8.2.4 broker 逐台升級:停 → 解鎖 → 裝新版 → 重新上鎖 → 啟動 → 健康才換下一台
{{
. ~/manual.env
for h in $BROKERS; do
  echo "== $h $(ssh $h.poc.lab "rpm -q --qf '%{VERSION}' confluent-server") -> $TO  ($(date +%T))"
  ssh $h.poc.lab "TO=$TO bash -s" <<'EOS'
set -e
sudo systemctl stop confluent-server
sudo dnf versionlock delete $(rpm -qa --qf '%{NAME}\n' 'confluent-*') >/dev/null 2>&1
sudo dnf -y -q --refresh --disablerepo='*' --enablerepo='cp-internal-*' --setopt=install_weak_deps=False install confluent-server-$TO confluent-security-$TO >/dev/null
for p in $(rpm -qa --qf '%{NAME}-%{VERSION}-%{RELEASE}\n' 'confluent-*'); do sudo dnf -q versionlock add "$p" >/dev/null; done
sudo systemctl start confluent-server
echo "  $(rpm -q confluent-server confluent-common | paste -sd' ') locks=$(sudo dnf -q versionlock list | grep -c confluent)"
EOS
  wait_for broker_ok $h || break
done
}}

##### 8.2.6 controller 逐台升級:follower 先、leader 最後;每台等 voter lag 0

  wait_for ctl_ok $h || break

##### 8.2.7 REST Proxy 逐台:先從 LB 摘除(down)→ 升級 → 健康 → 放回
{{
. ~/manual.env
lbconf() {  # <host to drain or empty>
  { echo 'upstream cp_rest {'; echo '    hash $remote_addr consistent;'
    for r in $RESTS; do printf '    server %s.poc.lab:8082 max_fails=2 fail_timeout=10s%s;\n' $r "$([ "$r" = "$1" ] && echo ' down')"; done
    echo '}'; echo 'upstream cp_c3 {'; echo '    server c3-01.poc.lab:9021 max_fails=2 fail_timeout=10s;'; echo '}'
    echo 'server { listen 8082; proxy_pass cp_rest; proxy_connect_timeout 3s; proxy_timeout 10m; }'
    echo 'server { listen 9021; proxy_pass cp_c3;   proxy_connect_timeout 3s; proxy_timeout 1h;  }'; } > $S/manual/cp.conf.new
  scp -q $S/manual/cp.conf.new lb01.poc.lab:/tmp/cp.conf.new
  ssh lb01.poc.lab 'sudo install -o root -g root -m 0644 /tmp/cp.conf.new /etc/nginx/stream.d/cp.conf && rm -f /tmp/cp.conf.new && sudo nginx -t -q && sudo nginx -s reload 2>/dev/null; sudo grep -c " down;" /etc/nginx/stream.d/cp.conf | sed "s/^/  servers marked down: /"'
}
for h in $RESTS; do
  echo "== drain $h ($(date +%T))"; lbconf $h; sleep 5
  ssh $h.poc.lab "TO=$TO bash -s" <<'EOS'
set -e
sudo systemctl stop confluent-kafka-rest
sudo dnf versionlock delete $(rpm -qa --qf '%{NAME}\n' 'confluent-*') >/dev/null 2>&1
sudo dnf -y -q --refresh --disablerepo='*' --enablerepo='cp-internal-*' --setopt=install_weak_deps=False install confluent-kafka-rest-$TO confluent-security-$TO >/dev/null
for p in $(rpm -qa --qf '%{NAME}-%{VERSION}-%{RELEASE}\n' 'confluent-*'); do sudo dnf -q versionlock add "$p" >/dev/null; done
sudo systemctl start confluent-kafka-rest
echo "  $(rpm -q confluent-kafka-rest)"
EOS
  rest_ok() { [ "$(curl -s -o /dev/null -w '%{http_code}' --cacert $PKI/ca.pem --cert $PKI/app-legacy-billing.crt --key $PKI/app-legacy-billing.key https://$h.poc.lab:8082/v3/clusters)" = 200 ]; }
  wait_for rest_ok || break
  echo "== undrain"; lbconf ""
done
}}
