{{
cat > ~/encrypt.sh <<'EOF'
# encrypt <dir> <plaintext file> <remote secrets path>   (manual 4.2.4 rule for which keys to protect)
encrypt() {
  local D=$1 F=$2 R=$3 KEYS
  KEYS=$(awk -F= '
    /^[[:space:]]*#/ || !/=/ { next }
    { k=$1; gsub(/[[:space:]]/,"",k); v=substr($0,index($0,"=")+1)
      if (v ~ /^[[:space:]]*$/ || v ~ /\$\{securepass:/) next
      if (k ~ /(password|credentials|passphrase|basic\.auth\.user\.info)$/) { print k; next }
      if (k ~ /sasl\.jaas\.config$/ && v ~ /password=/) { split(v,w,/[[:space:]]+/); m=(w[1]==""?w[2]:w[1]); print k "/" m "/password" } }' "$F" | paste -sd, -)
  install -m 0600 $S/secret-protection/master.secrets $D/$(basename $F .properties).security.properties
  CONFLUENT_SECURITY_MASTER_KEY=$(cat $S/secret-protection/master.key) HOME=$S/cli-home confluent secret file encrypt --config-file "$F" --local-secrets-file $D/$(basename $F .properties).security.properties --remote-secrets-file "$R" --config "$KEYS"
  echo "$(basename $D)/$(basename $F): keys=$(tr , '\n' <<<"$KEYS" | wc -l) securepass=$(grep -c securepass $F) plaintext-passwords=$(grep -ciE '(password|passphrase|credentials|user.info) *= *[^$ ]' $F)"
}
EOF
echo encrypt-helper-ready
}}
