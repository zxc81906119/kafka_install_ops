##### 4.2.1 Secret Protection master key(ops02,只做一次;passphrase 從 stdin 給,不放指令列)
{{
. ~/manual.env; set -a; . $S/secrets.env; set +a
install -d -m 0700 $S/secret-protection $S/cli-home
if test -s $S/secret-protection/master.key; then echo "已存在,不要重新產生"; else
  printf '%s' "$SECRET_MASTER_PASSPHRASE" | HOME=$S/cli-home confluent secret master-key generate --local-secrets-file $S/secret-protection/master.secrets --passphrase=- -o json | jq -r .master_key > $S/secret-protection/master.key
fi
chmod 0600 $S/secret-protection/master.key $S/secret-protection/master.secrets
ls -l $S/secret-protection/master.key $S/secret-protection/master.secrets; grep -o '^[^=]*' $S/secret-protection/master.secrets
}}

##### 4.2.4 加密(每台一次;要加密的 key 用手冊的 awk 算出)
{{
. ~/manual.env
for H in kc01 kc02 kc03 kb01 kb02 kb03 kb04; do
  D=$S/secret-protection/$H
  KEYS=$(awk -F= '
    /^[[:space:]]*#/ || !/=/ { next }
    { k=$1; gsub(/[[:space:]]/,"",k); v=substr($0,index($0,"=")+1)
      if (v ~ /^[[:space:]]*$/ || v ~ /\$\{securepass:/) next
      if (k ~ /(password|credentials|passphrase|basic\.auth\.user\.info)$/) { print k; next }
      if (k ~ /sasl\.jaas\.config$/ && v ~ /password=/) { split(v,w,/[[:space:]]+/); m=(w[1]==""?w[2]:w[1]); print k "/" m "/password" } }' \
    $D/server.properties | paste -sd, -)
  sha256sum $D/server.properties | awk '{print $1}' > $D/server.properties.plain.sha256
  install -m 0600 $S/secret-protection/master.secrets $D/kafka.security.properties
  CONFLUENT_SECURITY_MASTER_KEY=$(cat $S/secret-protection/master.key) HOME=$S/cli-home confluent secret file encrypt \
    --config-file $D/server.properties --local-secrets-file $D/kafka.security.properties \
    --remote-secrets-file /etc/confluent/kafka/security.properties --config "$KEYS"
  echo "$H: keys=$(tr , '\n' <<<"$KEYS" | wc -l) securepass=$(grep -c securepass $D/server.properties) plaintext-passwords=$(grep -ciE '(password|passphrase|credentials) *= *[^$ ]' $D/server.properties)"
done
}}
grep -E '^(config.providers|ssl.keystore.password|confluent.security.event.logger.exporter.kafka.sasl.jaas.config)' /var/lib/cp-installer/secret-protection/kb01/server.properties

##### 4.2.5 JMX exporter(設定只放加鹽 SHA-256;controller 用 kafka-internal.p12,broker 用 cp-server.p12)
{{
. ~/manual.env; set -a; . $S/secrets.env; set +a
JAR=jmx_prometheus_javaagent-1.0.1.jar
ls -l /srv/repo/raw/tools/$JAR
test -s $S/jmx-exporter.salt || (umask 077; openssl rand -hex 16 > $S/jmx-exporter.salt)
SALT=$(cat $S/jmx-exporter.salt)
HASH=$(printf '%s:%s' "$SALT" "$JMX_EXPORTER_PASSWORD" | sha256sum | awk '{print $1}')
for H in kc01 kc02 kc03 kb01 kb02 kb03 kb04; do
  case $H in kc*) KS=kafka-internal ;; *) KS=cp-server ;; esac
  Y=$S/secret-protection/$H/jmx-exporter.yml
  ( umask 077
    printf 'httpServer:\n  ssl:\n    keyStore:\n      filename: /etc/confluent/ssl/%s.p12\n      password: %s\n      type: PKCS12\n    certificate:\n      alias: %s\n  authentication:\n    basic:\n      username: prometheus\n      passwordHash: %s\n      algorithm: SHA-256\n      salt: %s\nlowercaseOutputName: true\nlowercaseOutputLabelNames: true\n' "$KS" "$STORE_PASSWORD" "$KS" "$HASH" "$SALT" > $Y
    cat $S/manual/ref/rules-kafka.yml >> $Y )
  scp -q /srv/repo/raw/tools/$JAR $Y $H.poc.lab:/tmp/
  ssh $H.poc.lab "sudo install -d -o root -g root -m 0755 /opt/jmx-exporter && sudo install -o root -g root -m 0644 /tmp/$JAR /opt/jmx-exporter/$JAR && sudo install -o root -g confluent -m 0640 /tmp/jmx-exporter.yml /etc/confluent/kafka/jmx-exporter.yml && sudo rm -f /tmp/$JAR /tmp/jmx-exporter.yml && echo \"\$(hostname): jmx ok ($KS)\""
done
}}
##### 4.2.6 systemd drop-in(controller heap 384m,broker 512m)+ 4.2.7 資料目錄
{{
for H in kc01 kc02 kc03 kb01 kb02 kb03 kb04; do
  case $H in kc*) HEAP=384m ;; *) HEAP=512m ;; esac
  cat > /tmp/10-cpctl.conf.$H <<EOF
# Managed by hand (manual 4.2.6) - drop-in for the RPM-provided confluent-server.service
[Service]
ExecStart=
ExecStart=/usr/bin/kafka-server-start /etc/confluent/kafka/server.properties
Environment="JAVA_HOME=/usr/lib/jvm/temurin-21-jdk"
Environment="KAFKA_HEAP_OPTS=-Xms$HEAP -Xmx$HEAP"
Environment="KAFKA_LOG4J_OPTS=-Dlog4j2.configurationFile=/etc/kafka/log4j2.yaml"
Environment="LOG_DIR=/var/log/kafka"
Environment="KAFKA_OPTS=-javaagent:/opt/jmx-exporter/jmx_prometheus_javaagent-1.0.1.jar=7071:/etc/confluent/kafka/jmx-exporter.yml"
EnvironmentFile=-/etc/confluent/kafka/service.env
LimitNOFILE=128000
TimeoutStopSec=180
Restart=on-failure
RestartSec=10
EOF
  scp -q /tmp/10-cpctl.conf.$H $H.poc.lab:/tmp/10-cpctl.conf && rm -f /tmp/10-cpctl.conf.$H
  ssh $H.poc.lab "sudo install -d -o root -g root -m 0755 /etc/systemd/system/confluent-server.service.d && sudo install -o root -g root -m 0644 /tmp/10-cpctl.conf /etc/systemd/system/confluent-server.service.d/10-cpctl.conf && sudo rm -f /tmp/10-cpctl.conf && sudo systemctl daemon-reload && sudo install -d -o cp-kafka -g confluent -m 0750 /var/lib/kafka/data && echo \"\$(hostname): dropin + data dir ok\""
done
}}
ssh kb01.poc.lab "systemctl cat confluent-server | grep -E '^# /|ExecStart=/|EnvironmentFile|HEAP'; sudo stat -c '%U:%G %a %n' /var/lib/kafka /var/lib/kafka/data"
