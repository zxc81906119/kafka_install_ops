# Secret Protection 加密過的設定檔是 "key = value"(等號兩邊有空白):sed 比對要容許空白
{{
for h in kc01 kc02 kc03; do
  ssh $h.poc.lab "sudo sed -i -E 's/^listener\.security\.protocol\.map *= *CONTROLLER:SSL$/listener.security.protocol.map = CONTROLLER:SSL,INTERNAL:SSL\ninter.broker.listener.name = INTERNAL/' /etc/confluent/kafka/server.properties"
  ssh $h.poc.lab "sudo grep -E '^(listener\.security\.protocol\.map|inter\.broker\.listener\.name) *=' /etc/confluent/kafka/server.properties" | sed "s/^/$h: /"
done
}}
