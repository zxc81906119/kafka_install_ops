##### 9.1 憑證輪替(同一張 CA 重簽 kafka-internal、cp-server;持續負載下滾動重啟)
install -m 0640 /var/lib/cp-installer/manual/rotate-lib.sh ~/rotate-lib.sh && echo ok
bash /var/lib/cp-installer/manual/load.sh start </dev/null >/var/lib/cp-installer/load/start.out 2>&1; cat /var/lib/cp-installer/load/start.out
{{
. ~/rotate-lib.sh
echo "== 換之前各端點出示的憑證"; served
for n in kafka-internal cp-server; do echo "$n expires $(openssl x509 -in $PKI/$n.crt -noout -enddate | cut -d= -f2)"; done
}}
{{
. ~/rotate-lib.sh
issue kafka-internal serverAuth,clientAuth $SANS_INTERNAL
issue cp-server serverAuth,clientAuth $SANS_SERVER
openssl verify -CAfile $PKI/ca.pem $PKI/kafka-internal.crt $PKI/cp-server.crt
ls -d $PKI/archive/*-kafka-internal $PKI/archive/*-cp-server | tail -2
}}
{{
. ~/rotate-lib.sh
push_all_ssl
push_ad
push_jump
}}
{{
. ~/rotate-lib.sh
restart_all
}}
{{
. ~/rotate-lib.sh
echo "== 換之後各端點出示的憑證(序號應全部是新的)"; served
echo "new kafka-internal $(openssl x509 -in $PKI/kafka-internal.crt -noout -serial)"; echo "new cp-server      $(openssl x509 -in $PKI/cp-server.crt -noout -serial)"
}}
bash /var/lib/cp-installer/manual/load.sh stop && bash /var/lib/cp-installer/manual/load.sh report
