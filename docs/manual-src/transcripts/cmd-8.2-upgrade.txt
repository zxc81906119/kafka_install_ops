##### 8.2 版本升級 CP 8.2.4 → 8.3.2(rpm;ops02,cpadmin)
{{
cat >> ~/manual.env <<'EOF'
export FROM=8.2.4 TO=8.3.2 TO_RELEASE=4.3
BROKERS="kb01 kb02 kb03 kb04" CONTROLLERS="kc01 kc02 kc03" RESTS="rest01 rest02"
# broker <host> 健康:已註冊且未 fence、沒有 under-replicated partition、MDS 有回應
broker_ok() { local id=$((100 + ${1#kb0})); kafka-broker-api-versions --bootstrap-server $B --command-config $CFG 2>/dev/null | grep -q "(id: $id rack: [a-z-]* isFenced: false)" && [ -z "$(kafka-topics --bootstrap-server $B --command-config $CFG --describe --under-replicated-partitions 2>/dev/null)" ] && curl -sf -o /dev/null --cacert $PKI/ca.pem https://$1.poc.lab:8090/v1/metadata/id; }
# controller <host> 健康:是 voter 且 lag 0
ctl_ok() { local id=${1#kc0}; kafka-metadata-quorum --bootstrap-controller $CB --command-config $CFG describe --replication 2>/dev/null | awk -v id=$id '$1==id && ($NF=="Leader"||$NF=="Follower") && $4+0==0 {ok=1} END {exit !ok}'; }
wait_for() { local i; for i in $(seq 1 60); do "$@" && { echo "  healthy after ~$((i*5))s"; return 0; }; sleep 5; done; echo "  NOT HEALTHY after 300s - stop here"; return 1; }
EOF
. ~/manual.env; echo "from $FROM to $TO (release $TO_RELEASE)"
}}
##### 8.2.1 升級前檢查(不改任何東西)
{{
. ~/manual.env
echo "== 目前版本"; for h in $BROKERS $CONTROLLERS; do echo "$h $(ssh $h.poc.lab "rpm -q --qf '%{VERSION}' confluent-server")"; done; for h in $RESTS; do echo "$h $(ssh $h.poc.lab "rpm -q --qf '%{VERSION}' confluent-kafka-rest")"; done; echo "c3-01 C3 $(ssh c3-01.poc.lab "rpm -q --qf '%{VERSION}' confluent-control-center-next-gen")"
echo "== 目標版本在內部 repo"; ssh kb01.poc.lab "sudo dnf -q --disablerepo='*' --enablerepo='cp-internal-confluent' list --showduplicates confluent-server confluent-security 2>/dev/null | grep -E '8\.3\.2|8\.2\.4'"
echo "== 8.3 已移除的靜態設定不可出現"; for h in $BROKERS $CONTROLLERS; do ssh $h.poc.lab "sudo grep -cE '^(message.format.version|message.downconversion.enable|message.timestamp.difference.max.ms)' /etc/confluent/kafka/server.properties" | sed "s/^/$h /"; done
echo "== 叢集健康"; for h in $BROKERS; do broker_ok $h && echo "$h ok" || echo "$h NOT OK"; done; for h in $CONTROLLERS; do ctl_ok $h && echo "$h ok" || echo "$h NOT OK"; done
echo "== 憑證效期"; echo "cp-server $(( ($(date -d "$(openssl x509 -enddate -noout -in $PKI/cp-server.crt | cut -d= -f2)" +%s) - $(date +%s)) / 86400 )) days, kafka-internal $(( ($(date -d "$(openssl x509 -enddate -noout -in $PKI/kafka-internal.crt | cut -d= -f2)" +%s) - $(date +%s)) / 86400 )) days"
echo "== 目前 finalize 的 metadata 版本"; kafka-features --bootstrap-server $B --command-config $CFG describe | grep -oE 'Feature: (confluent.metadata.version|metadata.version)[^E]*FinalizedVersionLevel: [^ ]+' | tr -s ' '
echo "== 相容性(官方相容表):C3 2.6.x 支援 CP 7.5–8.3;CP 8.3 支援 JDK 17/21/25;升級路徑 8.2 → 8.3 直接可行"
}}
##### 8.2.2 備份(設定檔、topic、SCRAM 使用者、features、ACL、RBAC 綁定、kraft.env)
{{
. ~/manual.env; set -a; . $S/secrets.env; set +a
BK=$S/upgrade/backup-$(date +%Y%m%d-%H%M%S); install -d -m 0700 $BK
for h in $CONTROLLERS $BROKERS $RESTS c3-01; do ssh $h.poc.lab "sudo tar -czf - -C / etc/confluent etc/systemd/system 2>/dev/null" > $BK/$h-etc.tgz; done
kafka-topics --bootstrap-server $B --command-config $CFG --describe > $BK/topics.txt 2>/dev/null
kafka-configs --bootstrap-server $B --command-config $CFG --describe --entity-type users > $BK/scram-users.txt 2>/dev/null
kafka-features --bootstrap-server $B --command-config $CFG describe > $BK/features.txt 2>/dev/null
kafka-acls --bootstrap-server $B --command-config $CFG --list > $BK/acls.txt 2>/dev/null
install -m 0600 /dev/null ~/.netrc-mds && printf 'machine kb01.poc.lab login svc-cpctl password %s\n' "$SVC_CPCTL_PASSWORD" > ~/.netrc-mds
for p in User:svc-c3 User:svc-rest User:confluent-audit User:opmenu-breakglass Group:kafka-admins Group:kafka-operators Group:kafka-devs User:app-orders User:app-payments User:app-legacy-billing; do
  curl -s --cacert $PKI/ca.pem --netrc-file ~/.netrc-mds -H 'Content-Type: application/json' -X POST -d "{\"clusters\":{\"kafka-cluster\":\"$CID\"}}" "$MDS/security/1.0/lookup/principal/$p/resources" > "$BK/rbac-${p/:/_}.json"
done; rm -f ~/.netrc-mds
cp $S/kraft.env $BK/; echo "$TO" > $BK/target; ln -sfn $BK $S/upgrade/last-backup
ls -la $BK | awk '{print $5, $9}' | column -t
}}
##### 8.2.3 啟動持續負載(SCRAM producer 帶序號 acks=all + 經 LB 的 REST v3,每秒一次)
install -m 0750 /var/lib/cp-installer/manual/ref/load.sh /var/lib/cp-installer/manual/load.sh && bash /var/lib/cp-installer/manual/load.sh start </dev/null >/var/lib/cp-installer/load/start.out 2>&1; cat /var/lib/cp-installer/load/start.out && sleep 20 && wc -l < /var/lib/cp-installer/load/produced.seq
##### 8.2.4 broker 逐台升級:停 → 解鎖 → 裝新版 → 重新上鎖 → 啟動 → 健康才換下一台
{{
. ~/manual.env
for h in $BROKERS; do
  echo "== $h $(ssh $h.poc.lab "rpm -q --qf '%{VERSION}' confluent-server") -> $TO  ($(date +%T))"
  ssh $h.poc.lab "TO=$TO bash -s" <<'EOS'
set -e
sudo systemctl stop confluent-server
sudo dnf versionlock delete $(rpm -qa --qf '%{NAME}\n' 'confluent-*') >/dev/null 2>&1
sudo dnf -y -q --refresh --disablerepo='*' --enablerepo='cp-internal-*' --setopt=install_weak_deps=False install confluent-server-$TO confluent-security-$TO >/dev/null
for p in $(rpm -qa --qf '%{NAME}-%{VERSION}-%{RELEASE}\n' 'confluent-*'); do sudo dnf -q versionlock add "$p" >/dev/null; done
sudo systemctl start confluent-server
echo "  $(rpm -q confluent-server confluent-common | paste -sd' ') locks=$(sudo dnf -q versionlock list | grep -c confluent)"
EOS
  wait_for broker_ok $h || break
done
}}
##### 8.2.5 回滾演練(finalize 前才可以):kb04 退回 8.2.4,再升回 8.3.2
{{
. ~/manual.env
kafka-features --bootstrap-server $B --command-config $CFG describe | grep -oE 'Feature: confluent.metadata.version[^E]*FinalizedVersionLevel: [^ ]+' | tr -s ' '
for v in $FROM $TO; do
  echo "== kb04 -> $v ($(date +%T))"
  ssh kb04.poc.lab "V=$v bash -s" <<'EOS'
set -e
sudo systemctl stop confluent-server
sudo dnf versionlock delete $(rpm -qa --qf '%{NAME}\n' 'confluent-*') >/dev/null 2>&1
D=(sudo dnf -y -q --refresh --disablerepo='*' --enablerepo='cp-internal-*' --setopt=install_weak_deps=False)
"${D[@]}" install confluent-server-$V confluent-security-$V >/dev/null 2>&1 || "${D[@]}" downgrade $(rpm -qa --qf '%{NAME}\n' 'confluent-*' | sed "s/\$/-$V/") >/dev/null
for p in $(rpm -qa --qf '%{NAME}-%{VERSION}-%{RELEASE}\n' 'confluent-*'); do sudo dnf -q versionlock add "$p" >/dev/null; done
sudo systemctl start confluent-server
echo "  $(rpm -q confluent-server confluent-common | paste -sd' ')"
EOS
  wait_for broker_ok kb04 || break
done
}}
##### 8.2.6 controller 逐台升級:follower 先、leader 最後;每台等 voter lag 0
{{
. ~/manual.env
L=$(kafka-metadata-quorum --bootstrap-controller $CB --command-config $CFG describe --status | awk '/LeaderId:/ {print $2}')
ORDER="$(for h in $CONTROLLERS; do [ "${h#kc0}" != "$L" ] && echo $h; done) kc0$L"
echo "leader is node $L; order: $(echo $ORDER)"
for h in $ORDER; do
  echo "== $h -> $TO ($(date +%T))"
  ssh $h.poc.lab "TO=$TO bash -s" <<'EOS'
set -e
sudo systemctl stop confluent-server
sudo dnf versionlock delete $(rpm -qa --qf '%{NAME}\n' 'confluent-*') >/dev/null 2>&1
sudo dnf -y -q --refresh --disablerepo='*' --enablerepo='cp-internal-*' --setopt=install_weak_deps=False install confluent-server-$TO confluent-security-$TO >/dev/null
for p in $(rpm -qa --qf '%{NAME}-%{VERSION}-%{RELEASE}\n' 'confluent-*'); do sudo dnf -q versionlock add "$p" >/dev/null; done
sudo systemctl start confluent-server
echo "  $(rpm -q confluent-server)"
EOS
  wait_for ctl_ok $h || break
done
kafka-metadata-quorum --bootstrap-controller $CB --command-config $CFG describe --status | grep -E 'LeaderId|MaxFollowerLag'
}}
##### 8.2.7 REST Proxy 逐台:先從 LB 摘除(down)→ 升級 → 健康 → 放回
{{
. ~/manual.env
lbconf() {  # <host to drain or empty>
  { echo 'upstream cp_rest {'; echo '    hash $remote_addr consistent;'
    for r in $RESTS; do printf '    server %s.poc.lab:8082 max_fails=2 fail_timeout=10s%s;\n' $r "$([ "$r" = "$1" ] && echo ' down')"; done
    echo '}'; echo 'upstream cp_c3 {'; echo '    server c3-01.poc.lab:9021 max_fails=2 fail_timeout=10s;'; echo '}'
    echo 'server { listen 8082; proxy_pass cp_rest; proxy_connect_timeout 3s; proxy_timeout 10m; }'
    echo 'server { listen 9021; proxy_pass cp_c3;   proxy_connect_timeout 3s; proxy_timeout 1h;  }'; } > $S/manual/cp.conf.new
  scp -q $S/manual/cp.conf.new lb01.poc.lab:/tmp/cp.conf.new
  ssh lb01.poc.lab 'sudo install -o root -g root -m 0644 /tmp/cp.conf.new /etc/nginx/stream.d/cp.conf && rm -f /tmp/cp.conf.new && sudo nginx -t -q && sudo nginx -s reload 2>/dev/null; sudo grep -c " down;" /etc/nginx/stream.d/cp.conf | sed "s/^/  servers marked down: /"'
}
for h in $RESTS; do
  echo "== drain $h ($(date +%T))"; lbconf $h; sleep 5
  ssh $h.poc.lab "TO=$TO bash -s" <<'EOS'
set -e
sudo systemctl stop confluent-kafka-rest
sudo dnf versionlock delete $(rpm -qa --qf '%{NAME}\n' 'confluent-*') >/dev/null 2>&1
sudo dnf -y -q --refresh --disablerepo='*' --enablerepo='cp-internal-*' --setopt=install_weak_deps=False install confluent-kafka-rest-$TO confluent-security-$TO >/dev/null
for p in $(rpm -qa --qf '%{NAME}-%{VERSION}-%{RELEASE}\n' 'confluent-*'); do sudo dnf -q versionlock add "$p" >/dev/null; done
sudo systemctl start confluent-kafka-rest
echo "  $(rpm -q confluent-kafka-rest)"
EOS
  rest_ok() { [ "$(curl -s -o /dev/null -w '%{http_code}' --cacert $PKI/ca.pem --cert $PKI/app-legacy-billing.crt --key $PKI/app-legacy-billing.key https://$h.poc.lab:8082/v3/clusters)" = 200 ]; }
  wait_for rest_ok || break
  echo "== undrain"; lbconf ""
done
}}
