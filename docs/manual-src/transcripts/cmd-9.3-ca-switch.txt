##### 9.3 CA 輪替 第 2 階段 switch:新 CA 開始簽章,舊 CA 只留在信任清單;所有憑證重簽
bash /var/lib/cp-installer/manual/load.sh start </dev/null >/var/lib/cp-installer/load/start.out 2>&1; cat /var/lib/cp-installer/load/start.out
{{
. ~/rotate-lib.sh
cd $PKI && umask 077
test -f ca-next.crt || { echo "沒有 ca-next:先做 add 階段"; exit 1; }
arch=archive/ca-$(date +%Y%m%d-%H%M%S); install -d -m 0700 $arch && cp -p ca.crt ca.key ca.srl $arch/ 2>/dev/null; ls $arch
mv ca.crt ca-old.crt && rm -f ca.key ca.srl
mv ca-next.crt ca.crt && mv ca-next.key ca.key
cat ca.crt ca-old.crt > ca.pem && chmod 0644 ca.pem
echo "signing CA now: $(openssl x509 -in ca.crt -noout -subject | sed 's/.*CN=\([^,]*\).*/\1/'); still trusted: $(openssl x509 -in ca-old.crt -noout -subject | sed 's/.*CN=\([^,]*\).*/\1/')"
}}
{{
. ~/rotate-lib.sh
issue kafka-internal serverAuth,clientAuth $SANS_INTERNAL
issue cp-server serverAuth,clientAuth $SANS_SERVER
issue app-legacy-billing clientAuth
issue opmenu-breakglass clientAuth
for n in kafka-internal cp-server app-legacy-billing opmenu-breakglass; do openssl verify -CAfile $PKI/ca.crt $PKI/$n.crt; done
}}
{{
. ~/rotate-lib.sh
truststores
push_all_ssl
trust_all
push_ad
push_jump
}}
{{
. ~/rotate-lib.sh
restart_all
}}
{{
. ~/rotate-lib.sh
echo "== 各端點出示的憑證(簽發者應全部是新 CA)"; served
echo "new CA: $(openssl x509 -in $PKI/ca.crt -noout -subject | sed 's/.*CN=\([^,]*\).*/\1/')"
echo "== app-legacy-billing 的新憑證要交給應用程式團隊:$PKI/app-legacy-billing.p12(舊憑證在 retire 之前仍可用)"
}}
bash /var/lib/cp-installer/manual/load.sh stop && bash /var/lib/cp-installer/manual/load.sh report
