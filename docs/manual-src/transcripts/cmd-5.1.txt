##### 5.1.1 SCRAM 使用者:用 --add-config-file(0600 暫存檔)讓密碼不出現在 ps
{{
. ~/manual.env; set -a; . $S/secrets.env; set +a
kafka-configs --bootstrap-server "$B" --command-config "$CFG" --describe --entity-type users
scram() {  # <user> <password>
  local f; f=$(mktemp); chmod 0600 "$f"
  printf 'SCRAM-SHA-512=[iterations=8192,password=%s]\n' "$2" > "$f"
  kafka-configs --bootstrap-server "$B" --command-config "$CFG" --alter --entity-type users --entity-name "$1" --add-config-file "$f"
  rm -f "$f"
}
scram app-orders "$APP_APP_ORDERS_PASSWORD"
scram app-payments "$APP_APP_PAYMENTS_PASSWORD"
scram confluent-audit "$AUDIT_PASSWORD"
kafka-configs --bootstrap-server "$B" --command-config "$CFG" --describe --entity-type users
}}

##### 5.1.3 RBAC 角色綁定(手冊完整清單,一行一筆)
{{

##### 5.1.3 驗證(MDS lookup;svc-cpctl 帳密放 0600 netrc)
{{
. ~/manual.env; set -a; . $S/secrets.env; set +a
install -m 0600 /dev/null ~/.netrc-mds && printf 'machine kb01.poc.lab login svc-cpctl password %s\n' "$SVC_CPCTL_PASSWORD" > ~/.netrc-mds
for p in User:app-orders Group:kafka-admins User:svc-rest User:confluent-audit User:svc-c3 User:app-legacy-billing; do
  curl -s --cacert $PKI/ca.pem --netrc-file ~/.netrc-mds -H 'Content-Type: application/json' -X POST -d "{\"clusters\":{\"kafka-cluster\":\"$CID\"}}" "$MDS/security/1.0/lookup/principal/$p/resources"; echo
done
rm -f ~/.netrc-mds
}}
