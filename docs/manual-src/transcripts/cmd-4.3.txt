##### 4.3.1 cluster id 與 directory id(ops02,只做一次)
{{
. ~/manual.env
test -f $S/kraft.env && echo "已存在,沿用" || {
  install -m 0600 /dev/null $S/kraft.env
  echo "CLUSTER_ID=$(kafka-storage random-uuid)" >> $S/kraft.env
  for h in kc01 kc02 kc03; do echo "DIR_ID_$h=$(kafka-storage random-uuid)" >> $S/kraft.env; done
}
. $S/kraft.env
INITIAL="1@kc01.poc.lab:9093:$DIR_ID_kc01,2@kc02.poc.lab:9093:$DIR_ID_kc02,3@kc03.poc.lab:9093:$DIR_ID_kc03"
echo "$CLUSTER_ID  $INITIAL"
}}
##### 4.3.2 格式化 controller(以 cp-kafka 執行;master key 由 root 讀 service.env 後 --preserve-env 傳給 cp-kafka;SCRAM 密碼來自 ops02 密碼檔)
{{
. ~/manual.env; set -a; . $S/secrets.env; . $S/kraft.env; set +a
INITIAL="1@kc01.poc.lab:9093:$DIR_ID_kc01,2@kc02.poc.lab:9093:$DIR_ID_kc02,3@kc03.poc.lab:9093:$DIR_ID_kc03"
for h in kc01 kc02 kc03; do
  echo "== $h: $(ssh $h.poc.lab 'sudo cat /var/lib/kafka/data/meta.properties 2>/dev/null || echo NOT-FORMATTED' | tail -1)"
  ssh $h.poc.lab "sudo bash -c \"set -a; . /etc/confluent/kafka/service.env; set +a; \
    sudo -u cp-kafka --preserve-env=CONFLUENT_SECURITY_MASTER_KEY env JAVA_HOME=/usr/lib/jvm/temurin-21-jdk KAFKA_HEAP_OPTS=-Xmx256m \
    KAFKA_OPTS=-Dlog4j2.configurationFile=/etc/kafka/tools-log4j2.yaml \
    /usr/bin/kafka-storage format --cluster-id $CLUSTER_ID --config /etc/confluent/kafka/server.properties \
      --initial-controllers '$INITIAL' \
      --add-scram 'SCRAM-SHA-512=[name=kafka-admin,password=$KAFKA_ADMIN_PASSWORD]' \
      --add-scram 'SCRAM-SHA-512=[name=confluent-audit,password=$AUDIT_PASSWORD]' \
    | grep -vi scram | tail -1\""
done
}}
##### 4.3.3 格式化 broker(observer)
{{
. ~/manual.env; set -a; . $S/kraft.env; set +a
for h in kb01 kb02 kb03 kb04; do
  ssh $h.poc.lab "sudo bash -c \"set -a; . /etc/confluent/kafka/service.env; set +a; \
    sudo -u cp-kafka --preserve-env=CONFLUENT_SECURITY_MASTER_KEY env JAVA_HOME=/usr/lib/jvm/temurin-21-jdk KAFKA_HEAP_OPTS=-Xmx256m \
    KAFKA_OPTS=-Dlog4j2.configurationFile=/etc/kafka/tools-log4j2.yaml \
    /usr/bin/kafka-storage format --cluster-id $CLUSTER_ID --config /etc/confluent/kafka/server.properties --no-initial-controllers | tail -1\""
done
}}
##### 4.3.4 驗證
{{
for h in kc01 kc02 kc03 kb01 kb02 kb03 kb04; do
  printf '%s ' $h; ssh $h.poc.lab "sudo sed -n 's/^\(cluster.id\|node.id\|directory.id\)=//p' /var/lib/kafka/data/meta.properties | paste -sd' '; sudo stat -c '%U' /var/lib/kafka/data/meta.properties"
done | paste - -
grep DIR_ID /var/lib/cp-installer/kraft.env
}}
