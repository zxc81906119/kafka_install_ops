##### 2.2 外網下載:RPM(下載機 = 與節點相同 base 的乾淨 Rocky 9,root)
mkdir -p /tmp/stage /out/gpg-keys /out/rpm/confluent /out/rpm/adoptium /out/rpm/os-deps /out/bootstrap
sed -i 's/^enabled=0/enabled=1/' /etc/yum.repos.d/rocky*.repo && dnf -q repolist
printf '[confluent-8.3]\nname=Confluent 8.3\nbaseurl=https://packages.confluent.io/rpm/8.3\ngpgcheck=1\ngpgkey=https://packages.confluent.io/rpm/8.3/archive.key\nenabled=1\n[confluent-c3]\nname=Confluent Control Center next-gen\nbaseurl=https://packages.confluent.io/confluent-control-center-next-gen/rpm/\ngpgcheck=1\ngpgkey=https://packages.confluent.io/confluent-control-center-next-gen/rpm/archive.key\nenabled=1\n[adoptium]\nname=Adoptium\nbaseurl=https://packages.adoptium.net/artifactory/rpm/rhel/9/x86_64\ngpgcheck=1\ngpgkey=https://packages.adoptium.net/artifactory/api/gpg/key/public\nenabled=1\n' > /etc/yum.repos.d/upstream-confluent.repo && cat /etc/yum.repos.d/upstream-confluent.repo
curl -fsSL https://packages.confluent.io/rpm/8.3/archive.key -o /out/gpg-keys/confluent.asc && curl -fsSL https://packages.confluent.io/confluent-control-center-next-gen/rpm/archive.key -o /out/gpg-keys/confluent-c3.asc && curl -fsSL https://packages.adoptium.net/artifactory/api/gpg/key/public -o /out/gpg-keys/adoptium.asc && cp /etc/pki/rpm-gpg/RPM-GPG-KEY-Rocky-9 /out/gpg-keys/rocky-9.asc && ls -l /out/gpg-keys
for k in /out/gpg-keys/*.asc; do rpm --import "$k"; done; rpm -q gpg-pubkey --qf '%{SUMMARY}\n'
dnf -q makecache && dnf -q repolist
dnf -q download --resolve --arch x86_64 --arch noarch --destdir /tmp/stage confluent-server-8.3.2 confluent-security-8.3.2 confluent-kafka-rest-8.3.2 confluent-cli-8.3.2 && ls /tmp/stage | wc -l
dnf -q download --resolve --arch x86_64 --arch noarch --destdir /tmp/stage confluent-control-center-next-gen-2.6.1 cyrus-sasl openssl-devel && ls /tmp/stage | wc -l
dnf -q repoquery --repo adoptium --qf '%{version}-%{release}' temurin-21-jdk | sort -V | tail -3
dnf -q download --resolve --arch x86_64 --arch noarch --destdir /tmp/stage temurin-21-jdk-21.0.12.1.0.1 && ls /tmp/stage | wc -l
dnf -q download --resolve --arch x86_64 --arch noarch --destdir /tmp/stage chrony firewalld && ls /tmp/stage | wc -l
dnf -q repoquery --repo confluent-8.3 --repo confluent-c3 --qf '%{name}' | sort -u > /tmp/cp-names && dnf -q repoquery --repo adoptium --qf '%{name}' | sort -u > /tmp/ad-names && wc -l /tmp/cp-names /tmp/ad-names
for f in /tmp/stage/*.rpm; do n=$(rpm -qp --qf '%{NAME}' "$f" 2>/dev/null); if grep -qx "$n" /tmp/cp-names; then d=confluent; elif grep -qx "$n" /tmp/ad-names; then d=adoptium; else d=os-deps; fi; [ -f "/out/rpm/$d/$(basename "$f")" ] || mv "$f" "/out/rpm/$d/"; done; for d in confluent adoptium os-deps; do echo "$d $(ls /out/rpm/$d | wc -l)"; done
ls /out/rpm/confluent /out/rpm/adoptium
find /out/rpm -name '*.rpm' -print0 | xargs -0 rpm -K | grep -vc 'digests signatures OK'
