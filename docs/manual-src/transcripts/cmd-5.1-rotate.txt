##### 改了 confluent-audit 密碼 → controller/broker 設定要重新產生、加密、派送,再逐台滾動重啟(手冊 4.2.3、4.2.4、4.2.8)
{{
. ~/manual.env
G=/var/lib/cp-installer/manual/gen-server-properties.sh
bash $G kc01 controller 1; bash $G kc02 controller 2; bash $G kc03 controller 3
bash $G kb01 broker 101 rack-a; bash $G kb02 broker 102 rack-a; bash $G kb03 broker 103 rack-b; bash $G kb04 broker 104 rack-b
for H in kc01 kc02 kc03 kb01 kb02 kb03 kb04; do
  D=$S/secret-protection/$H
  KEYS=$(awk -F= '
    /^[[:space:]]*#/ || !/=/ { next }
    { k=$1; gsub(/[[:space:]]/,"",k); v=substr($0,index($0,"=")+1)
      if (v ~ /^[[:space:]]*$/ || v ~ /\$\{securepass:/) next
      if (k ~ /(password|credentials|passphrase|basic\.auth\.user\.info)$/) { print k; next }
      if (k ~ /sasl\.jaas\.config$/ && v ~ /password=/) { split(v,w,/[[:space:]]+/); m=(w[1]==""?w[2]:w[1]); print k "/" m "/password" } }' \
    $D/server.properties | paste -sd, -)
  sha256sum $D/server.properties | awk '{print $1}' > $D/server.properties.plain.sha256
  install -m 0600 $S/secret-protection/master.secrets $D/kafka.security.properties
  CONFLUENT_SECURITY_MASTER_KEY=$(cat $S/secret-protection/master.key) HOME=$S/cli-home confluent secret file encrypt --config-file $D/server.properties --local-secrets-file $D/kafka.security.properties --remote-secrets-file /etc/confluent/kafka/security.properties --config "$KEYS"
  ssh $H.poc.lab "mkdir -p /tmp/cpcfg && chmod 0700 /tmp/cpcfg"
  scp -q $D/server.properties $D/kafka.security.properties $H.poc.lab:/tmp/cpcfg/
  ssh $H.poc.lab "sudo install -o root -g confluent -m 0640 /tmp/cpcfg/server.properties /etc/confluent/kafka/server.properties && sudo install -o root -g confluent -m 0640 /tmp/cpcfg/kafka.security.properties /etc/confluent/kafka/security.properties && sudo rm -rf /tmp/cpcfg && echo \"\$(hostname): config updated\""
done
}}
##### 4.2.8 滾動重啟:一次一台,controller → broker;每台等健康(controller:voter lag 0;broker:未 fence 且 under-replicated 為空)
{{
. ~/manual.env
for h in kc01 kc02 kc03; do
  id=${h#kc0}; ssh $h.poc.lab "sudo systemctl restart confluent-server"
  for i in $(seq 1 60); do kafka-metadata-quorum --bootstrap-controller $CB --command-config $CFG describe --replication 2>/dev/null | awk -v id=$id '$1==id && ($NF=="Leader"||$NF=="Follower") && $4+0==0 {ok=1} END {exit !ok}' && break; sleep 5; done
  echo "$h restarted, voter $id caught up after ~$((i*5))s"
done
for h in kb01 kb02 kb03 kb04; do
  id=$((100 + ${h#kb0})); ssh $h.poc.lab "sudo systemctl restart confluent-server"
  for i in $(seq 1 60); do sleep 5; kafka-broker-api-versions --bootstrap-server $B --command-config $CFG 2>/dev/null | grep -q "(id: $id rack: [a-z-]* isFenced: false)" || continue; [ -z "$(kafka-topics --bootstrap-server $B --command-config $CFG --describe --under-replicated-partitions 2>/dev/null)" ] && break; done
  echo "$h restarted, broker $id healthy after ~$((i*5))s"
done
}}
