##### #43 fail closed:先建立故障設定並確認生效,才執行 item 21(重啟 broker)
{{
. ~/opmenu-test.sh
ssh jump01.poc.lab "printf '. /etc/opmenu/opmenu-test.conf\nOPMENU_BOOTSTRAP=nowhere.invalid:9096\n' | sudo -n tee /tmp/opmenu-broken.conf >/dev/null; sudo -n chmod 0644 /tmp/opmenu-broken.conf; sudo -n -u opuser bash -c '. /tmp/opmenu-broken.conf; echo effective OPMENU_BOOTSTRAP=\$OPMENU_BOOTSTRAP'"
before=$(ssh kb01.poc.lab "systemctl show confluent-server -p ActiveEnterTimestamp --value")
out=$(CONF=/tmp/opmenu-broken.conf R alice "$HUMAN_DEFAULT_PASSWORD" 21 kb01.poc.lab:confluent-server)
echo "$out" | grep -E '無法檢查|已重啟' | head -2
after=$(ssh kb01.poc.lab "systemctl show confluent-server -p ActiveEnterTimestamp --value")
echo "#43 kb01 start time before=[$before] after=[$after] $( [ "$before" = "$after" ] && echo 'NOT restarted (fail closed)' || echo 'RESTARTED!')"
ssh jump01.poc.lab 'sudo -n rm -f /tmp/opmenu-broken.conf'
}}
