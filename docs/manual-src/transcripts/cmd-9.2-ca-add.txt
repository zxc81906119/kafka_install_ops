{{
. ~/rotate-lib.sh
cd $PKI && umask 077
test -f ca-next.crt && echo "ca-next 已存在,沿用" || {
  openssl genrsa -out ca-next.key 3072 2>/dev/null
  openssl req -x509 -new -key ca-next.key -sha256 -days 3650 -out ca-next.crt -subj "/CN=PoC Internal CA $(date +%Y%m%d%H%M)/OU=Confluent/O=PoC Bank/C=TW" -addext "basicConstraints=critical,CA:TRUE" -addext "keyUsage=critical,keyCertSign,cRLSign"
}
cat ca.crt ca-next.crt > ca.pem && chmod 0644 ca.pem
for f in ca.crt ca-next.crt; do echo "$f: $(openssl x509 -in $f -noout -subject | sed 's/.*CN=\([^,]*\).*/\1/') fp=$(openssl x509 -in $f -noout -fingerprint -sha256 | cut -d= -f2 | tr -d : | cut -c1-16)"; done
echo "bundle: $(grep -c 'BEGIN CERT' ca.pem) CAs"
}}
